local function Lerp(a, b, t) return a + (b - a) * t end

local function LerpColor(c1, c2, t)
    return Color3.new(Lerp(c1.R, c2.R, t), Lerp(c1.G, c2.G, t), Lerp(c1.B, c2.B, t))
end

local function Create(type, props)
    local drawing = Drawing.new(type)
    for k, v in pairs(props) do drawing[k] = v end
    table.insert(Library.Drawings, drawing)
    return drawing
end

local function IsMouseOver(pos, size)
    local mPos = UIS:GetMouseLocation()
    return mPos.X >= pos.X and mPos.X <= pos.X + size.X 
       and mPos.Y >= pos.Y and mPos.Y <= pos.Y + size.Y
end

local function RGBtoHSV(color)
    local r, g, b = color.R, color.G, color.B
    local min, max = math.min(r, g, b), math.max(r, g, b)
    local delta = max - min
    local h, s, v
    v = max
    if delta == 0 then
        h = 0
        s = 0
    else
        s = delta / max
        local dr = (((max - r) / 6) + (delta / 2)) / delta
        local dg = (((max - g) / 6) + (delta / 2)) / delta
        local db = (((max - b) / 6) + (delta / 2)) / delta
        if r == max then h = db - dg
        elseif g == max then h = (1 / 3) + dr - db
        else h = (2 / 3) + dg - dr end
        if h < 0 then h = h + 1 end
        if h > 1 then h = h - 1 end
    end
    return h, s, v
end

function Library:SaveConfig(filename)
    if not writefile then return false end
    local config = {}
    for flag, value in pairs(Library.Flags) do
        if typeof(value) == "Color3" then
            config[flag] = {value.R, value.G, value.B}
        elseif typeof(value) == "EnumItem" then
            config[flag] = {value.EnumType.Name, value.Name}
        else
            config[flag] = value
        end
    end
    local success, encoded = pcall(function()
        return HttpService:JSONEncode(config)
    end)
    if success then
        writefile((filename or "SilentHub_Config") .. ".json", encoded)
        return true
    end
    return false
end

function Library:LoadConfig(filename)
    if not readfile or not isfile then return false end
    local file = (filename or "SilentHub_Config") .. ".json"
    if not isfile(file) then return false end
    local success, decoded = pcall(function()
        return HttpService:JSONDecode(readfile(file))
    end)
    if success and decoded then
        for flag, value in pairs(decoded) do
            if Library.Flags[flag] ~= nil then
                if typeof(Library.Flags[flag]) == "Color3" and type(value) == "table" then
                    Library.Flags[flag] = Color3.new(value[1], value[2], value[3])
                elseif typeof(Library.Flags[flag]) == "EnumItem" and type(value) == "table" then
                    local enumType = Enum[value[1]]
                    if enumType and enumType[value[2]] then
                        Library.Flags[flag] = enumType[value[2]]
                    end
                else
                    Library.Flags[flag] = value
                end
            end
        end
        return true
    end
    return false
end

local function UnloadUI()
    CAS:UnbindAction("SilentUI_ScrollSink")
    
    for _, c in pairs(Library.Connections) do c:Disconnect() end
    for _, d in pairs(Library.Drawings) do d:Remove() end
    
    if Library.Watermark then
        if Library.Watermark.Bg then Library.Watermark.Bg:Remove() end
        if Library.Watermark.Txt then Library.Watermark.Txt:Remove() end
    end

    Library.Drawings = {}
    Library.Connections = {}
    Library.Flags = {}
    getgenv().SilentUI_Unload = nil
end
getgenv().SilentUI_Unload = UnloadUI

function Library:Rejoin()
    if #Players:GetPlayers() <= 1 then
        LocalPlayer:Kick("\nRejoining...")
        task.wait(1)
        TeleportService:Teleport(game.PlaceId, LocalPlayer)
    else
        TeleportService:TeleportToPlaceInstance(game.PlaceId, game.JobId, LocalPlayer)
    end
end

    local showWm = Library.Flags["WatermarkToggle"] ~= false
    local targetAlpha = (showWm and Library.Open) and 1 or 0
    WatermarkAlpha = Lerp(WatermarkAlpha, targetAlpha, 15 * dt)

    if WatermarkAlpha < 0.01 then
        Library.Watermark.Bg.Visible = false
        Library.Watermark.Txt.Visible = false
        return
    end

    local now = tick()
    if now - LastFpsUpdate > 0.5 then
        CachedFps = math.floor(1 / dt)
        LastFpsUpdate = now
    end

    local parts = {}
    
    if Library.Flags["Watermark_GameName"] ~= false then
        local success, name = pcall(function() return MarketplaceService:GetProductInfo(game.PlaceId).Name end)
        table.insert(parts, {Text = success and name or "Roblox", Flag = "Watermark_Color_GameName"})
    end

    if Library.Flags["Watermark_Fps"] ~= false then
        table.insert(parts, {Text = CachedFps .. " fps", Flag = "Watermark_Color_Fps"})
    end

    if Library.Flags["Watermark_Ping"] ~= false then
        local ping = 0
        pcall(function()
            ping = math.floor(LocalPlayer:GetNetworkPing() * 1000)
        end)
        table.insert(parts, {Text = ping .. "ms", Flag = "Watermark_Color_Ping"})
    end

    if Library.Flags["Watermark_PlayerCount"] ~= false then
        table.insert(parts, {Text = #Players:GetPlayers() .. "/" .. Players.MaxPlayers, Flag = "Watermark_Color_PlayerCount"})
    end

    if Library.Flags["Watermark_Time"] ~= false then
        table.insert(parts, {Text = os.date("%H:%M:%S"), Flag = "Watermark_Color_Time"})
    end

    local fullStr = table.concat(parts, " | ")
    if #parts == 0 then
        fullStr = "SilentHub"
    end

    Library.Watermark.Txt.Text = fullStr
    Library.Watermark.Txt.Transparency = WatermarkAlpha
    Library.Watermark.Txt.Font = Library.Theme.Font

    local vp = workspace.CurrentCamera.ViewportSize
    local wmSize = Library.Watermark.Txt.TextBounds
    Library.Watermark.Bg.Size = wmSize + Vector2.new(10, 4)

    local targetX = math.floor(vp.X - wmSize.X - 25)
    local targetY = 20
    local pos = Vector2.new(targetX, targetY)

    Library.Watermark.Bg.Position = pos
    Library.Watermark.Bg.Transparency = 0.7 * WatermarkAlpha
    Library.Watermark.Bg.Visible = true
    Library.Watermark.Txt.Visible = true
    Library.Watermark.Txt.Position = pos + Vector2.new(5, 2)
end))

Library.Notifications = {}

function Library:Notify(text, duration)
    local vp = workspace.CurrentCamera.ViewportSize
    
    local note = {
        Text = text, 
        Duration = duration or 3, 
        SpawnTime = tick(),
        Exiting = false,
        X = vp.X, 
        Drawings = {}
    }
    
    note.Drawings.Txt = Create("Text", {Font=Library.Theme.Font, Size=13, ZIndex=202, Text=text, Visible=true})
    note.Drawings.Box = Create("Square", {Filled=true, ZIndex=200, Visible=true})
    note.Drawings.Out = Create("Square", {Filled=false, Thickness=1, ZIndex=201, Visible=true})

    table.insert(Library.Notifications, note)
end

table.insert(Library.Connections, RunService.RenderStepped:Connect(function(dt)
    local vp = workspace.CurrentCamera.ViewportSize
    local noteCount = #Library.Notifications
    local startY = (vp.Y / 2) - (noteCount * 34 / 2) 

    for i = noteCount, 1, -1 do
        local note = Library.Notifications[i]
        
        if tick() - note.SpawnTime > note.Duration then
            note.Exiting = true
        end

        local width = note.Drawings.Txt.TextBounds.X + 20
        local targetX = note.Exiting and (vp.X + 50) or (vp.X - width - 20)
        
        note.X = Lerp(note.X, targetX, 10 * dt)
        
        local yPos = startY + ((i - 1) * 34)
        local pos = Vector2.new(note.X, yPos)
        local size = Vector2.new(width, 24)

        note.Drawings.Box.Position = pos
        note.Drawings.Box.Size = size
        note.Drawings.Box.Color = Library.Theme.Main

        note.Drawings.Out.Position = pos
        note.Drawings.Out.Size = size
        note.Drawings.Out.Color = Library.Theme.Accent

        note.Drawings.Txt.Position = pos + Vector2.new(10, 5)
        note.Drawings.Txt.Color = Library.Theme.Text
        note.Drawings.Txt.Font = Library.Theme.Font

        if note.Exiting and math.abs(note.X - targetX) < 5 then
            note.Drawings.Box:Remove()
            note.Drawings.Out:Remove()
            note.Drawings.Txt:Remove()
            table.remove(Library.Notifications, i)
        end
    end
end))

local ActiveTooltip = {
    Text = nil,
    Txt = Create("Text", {Size=11, ZIndex=5000, Visible=false}),

local function ShowTooltip(text)
    if not text then 
        ActiveTooltip.Txt.Visible = false
        ActiveTooltip.Bg.Visible = false
        ActiveTooltip.Out.Visible = false
        ActiveTooltip.Active = false
        return 
    end
    ActiveTooltip.Active = true
    local mPos = UIS:GetMouseLocation() + Vector2.new(15, 15)
    ActiveTooltip.Txt.Text = text
    ActiveTooltip.Txt.Font = Library.Theme.Font
    ActiveTooltip.Txt.Color = Library.Theme.Text
    ActiveTooltip.Txt.Position = mPos + Vector2.new(5, 3)
    ActiveTooltip.Txt.Visible = true

    local size = ActiveTooltip.Txt.TextBounds + Vector2.new(10, 6)
    ActiveTooltip.Bg.Position = mPos
    ActiveTooltip.Bg.Size = size
    ActiveTooltip.Bg.Color = Library.Theme.Sidebar
    ActiveTooltip.Bg.Visible = true

    ActiveTooltip.Out.Position = mPos
    ActiveTooltip.Out.Size = size
    ActiveTooltip.Out.Color = Library.Theme.Outline
    ActiveTooltip.Out.Visible = true
end

function Library:Window(options)
    options = options or {}
    local Win = {
        Title = options.Title or "SilentHub",
        Size = options.Size or Vector2.new(600, 450),
        Position = options.Position or Vector2.new(300, 300),
        Tabs = {},
        ActiveTab = nil,
        Dragging = false,
        DragOffset = Vector2.new(0,0),
        OpenedPopup = nil,
        KeybindToChange = nil,
        TextBoxFocused = nil
    }
    Library.CurrentWindow = Win

    local MouseClicked = false

    CAS:BindActionAtPriority("SilentUI_ScrollSink", function(actionName, inputState, inputObj)
        if Library.Open and Win then
            if IsMouseOver(Win.Position, Win.Size) then
                return Enum.ContextActionResult.Sink
            end
        end
        return Enum.ContextActionResult.Pass
    end, false, 3000, Enum.UserInputType.MouseWheel)

    local MainOutline = Create("Square", {Thickness=1, Filled=false, ZIndex=1})
    local MainBg = Create("Square", {Filled=true, ZIndex=0})
    local SidebarBg = Create("Square", {Filled=true, ZIndex=1})
    local SidebarOutline = Create("Square", {Thickness=1, Filled=false, ZIndex=1})
    local TitleText = Create("Text", {Text=Win.Title, Size=16, Font=Library.Theme.Font, ZIndex=2, Center=false})
    local TitleAccent = Create("Square", {Filled=true, ZIndex=2})

    table.insert(Library.Connections, UIS.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseWheel and Library.Open and Win.ActiveTab and Win.ActiveTab.ActiveSubTab then
            if IsMouseOver(Win.Position, Win.Size) and not Win.OpenedPopup then
                local sub = Win.ActiveTab.ActiveSubTab
                local visibleHeight = Win.Size.Y - 55 
                
                if sub.ContentHeight > visibleHeight then
                    local scrollSpeed = 40
                    sub.TargetScroll = sub.TargetScroll - (input.Position.Z * scrollSpeed)
                    local maxScroll = math.max(0, sub.ContentHeight - visibleHeight + 10)
                    sub.TargetScroll = math.clamp(sub.TargetScroll, 0, maxScroll)
                end
            end
        end
    end))

    table.insert(Library.Connections, UIS.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            MouseClicked = true
            
            if Library.Open and IsMouseOver(Win.Position, Vector2.new(140, 40)) and not Win.OpenedPopup then
                Win.Dragging = true
                Win.DragOffset = UIS:GetMouseLocation() - Win.Position
            end

            if Win.OpenedPopup then
                local pObj = Win.OpenedPopup
                local pBounds = pObj.PopupBounds or Vector2.new(0,0)
                local pSize = pObj.PopupSize or Vector2.new(0,0)
                local bPos = pObj.ButtonPos or Vector2.new(0,0)
                local bSize = pObj.ButtonSize or Vector2.new(0,0)

                if not IsMouseOver(pBounds, pSize) and not IsMouseOver(bPos, bSize) then
                    pObj:Close()
                end
            end

            if Win.TextBoxFocused then
                Win.TextBoxFocused.Focused = false
                Win.TextBoxFocused = nil
            end

        elseif input.KeyCode == Enum.KeyCode.RightControl then 
            Library.Open = not Library.Open
        elseif input.KeyCode == Enum.KeyCode.End then
             UnloadUI()
        end

        if Win.KeybindToChange then
            if input.KeyCode == Enum.KeyCode.Escape then
                Win.KeybindToChange.Binding = false
                Win.KeybindToChange = nil
                Library:Notify("Keybind binding cancelled")
                return
            elseif input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode ~= Enum.KeyCode.Unknown then
                local kb = Win.KeybindToChange
                kb.Value = input.KeyCode
                if kb.Flag then Library.Flags[kb.Flag] = kb.Value end
                kb.Binding = false
                Win.KeybindToChange = nil
                Library:Notify("Keybind set to " .. tostring(input.KeyCode.Name))
                return
            end
        end

        if not Win.KeybindToChange then
            for _, tab in pairs(Win.Tabs) do
                for _, sub in pairs(tab.SubTabs) do
                    for _, el in pairs(sub.Elements) do
                        if el.Class == "Keybind" and el.Value == input.KeyCode then
                            if el.Mode == "Toggle" then
                                el.Toggled = not el.Toggled
                                el.Callback(el.Toggled)
                            elseif el.Mode == "Hold" then
                                el.Toggled = true
                                el.Callback(true)
                            end
                        end
                    end
                end
            end
        end
    end))

    table.insert(Library.Connections, UIS.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Keyboard and Win.TextBoxFocused then
            local tb = Win.TextBoxFocused
            if input.KeyCode == Enum.KeyCode.BackSpace then
                if #tb.Value > 0 then
                    tb.Value = tb.Value:sub(1, #tb.Value - 1)
                    if tb.Callback then tb.Callback(tb.Value) end
                end
            elseif input.KeyCode == Enum.KeyCode.Return or input.KeyCode == Enum.KeyCode.Escape then
                tb.Focused = false
                Win.TextBoxFocused = nil
            else
                local char = input.KeyCode.Name
                local shift = UIS:IsKeyDown(Enum.KeyCode.LeftShift) or UIS:IsKeyDown(Enum.KeyCode.RightShift)
                
                if #char == 1 then
                    tb.Value = tb.Value .. (shift and char:upper() or char:lower())
                    if tb.Callback then tb.Callback(tb.Value) end
                elseif input.KeyCode == Enum.KeyCode.Space then
                    tb.Value = tb.Value .. " "
                    if tb.Callback then tb.Callback(tb.Value) end
                end
            end
        end
    end))

    table.insert(Library.Connections, UIS.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then Win.Dragging = false end
        
        if not Win.KeybindToChange and input.UserInputType == Enum.UserInputType.Keyboard then
            for _, tab in pairs(Win.Tabs) do
                for _, sub in pairs(tab.SubTabs) do
                    for _, el in pairs(sub.Elements) do
                        if el.Class == "Keybind" and el.Value == input.KeyCode and el.Mode == "Hold" then
                            el.Toggled = false
                            el.Callback(false)
                        end
                    end
                end
            end
        end
    end))

    table.insert(Library.Connections, RunService.RenderStepped:Connect(function(dt)
        local targetAlpha = Library.Open and 1 or 0
        Library.Alpha = Lerp(Library.Alpha, targetAlpha, 15 * dt)

        if ActiveTooltip.Active then
            ShowTooltip(nil)
        end

        if Library.Alpha < 0.01 then 
            for _, d in pairs(Library.Drawings) do d.Visible = false end
            ShowTooltip(nil)
            MouseClicked = false
            return 
        end

        if Win.Dragging then
            local targetPos = UIS:GetMouseLocation() - Win.DragOffset
            Win.Position = Vector2.new(Lerp(Win.Position.X, targetPos.X, 20*dt), Lerp(Win.Position.Y, targetPos.Y, 20*dt))
        end

        local p, s, a = Win.Position, Win.Size, Library.Alpha

        MainBg.Visible = true;
        MainBg.Position = p; MainBg.Size = s; MainBg.Color = Library.Theme.Main; MainBg.Transparency = a
        MainOutline.Visible = true;
        MainOutline.Position = p; MainOutline.Size = s; MainOutline.Color = Library.Theme.Outline; MainOutline.Transparency = a

        local sbW = 140
        SidebarBg.Visible = true;
        SidebarBg.Position = p; SidebarBg.Size = Vector2.new(sbW, s.Y); SidebarBg.Color = Library.Theme.Sidebar;
        SidebarBg.Transparency = a
        SidebarOutline.Visible = true; SidebarOutline.Position = p; SidebarOutline.Size = Vector2.new(sbW, s.Y);
        SidebarOutline.Color = Library.Theme.Outline; SidebarOutline.Transparency = a

        TitleText.Visible = true;
        TitleText.Position = p + Vector2.new(15, 12); 
        TitleText.Color = Library.Theme.Accent; 
        TitleText.Transparency = a
        TitleText.Font = Library.Theme.Font
        
        TitleAccent.Visible = true;
        TitleAccent.Position = p + Vector2.new(0, 40); TitleAccent.Size = Vector2.new(sbW, 1); TitleAccent.Color = Library.Theme.Accent;
        TitleAccent.Transparency = a

        local tabY = 50
        for _, tab in ipairs(Win.Tabs) do
            tab:Render(p + Vector2.new(0, tabY), sbW, dt, a)
            tabY = tabY + 30
        end

        if Win.ActiveTab then
            local startX = p.X + sbW + 15
            local topY = p.Y + 12
            local currentX = startX

            if Win.ActiveTab.SubTabs then
                for _, sub in ipairs(Win.ActiveTab.SubTabs) do
                    sub:RenderHeader(Vector2.new(currentX, topY), dt, a)
                    currentX = currentX + sub.Width + 15
                end
            end

            if Win.ActiveTab.ActiveSubTab then
                local sub = Win.ActiveTab.ActiveSubTab
                
                local totalHeight = 0
                for _, el in ipairs(sub.Elements) do
                    totalHeight = totalHeight + el:GetHeight() + 5
                end
                sub.ContentHeight = totalHeight

                sub.ScrollY = Lerp(sub.ScrollY, sub.TargetScroll, 15 * dt)

                local viewX = startX
                local viewY = p.Y + 45
                local viewWidth = s.X - sbW - 30
                local viewHeight = s.Y - 55
                local clipTop = viewY
                local clipBottom = viewY + viewHeight

                local currentY = viewY - sub.ScrollY

                for _, el in ipairs(sub.Elements) do
                    local elHeight = el:GetHeight()
                    local elPos = Vector2.new(viewX, currentY)
                    
                    if (currentY >= clipTop - 5) and (currentY + elHeight <= clipBottom + 5) then
                        el:Render(elPos, viewWidth, dt, a, Win.OpenedPopup, Win.Dragging)
                    else
                        el:Hide()
                    end
                    
                    currentY = currentY + elHeight + 5
                end

                local visibleHeight = viewHeight
                if sub.ContentHeight > visibleHeight then
                    local trackHeight = visibleHeight - 10
                    local handleHeight = math.max(20, (visibleHeight / sub.ContentHeight) * trackHeight)
                    local maxScroll = sub.ContentHeight - visibleHeight
                    local scrollPct = maxScroll > 0 and (sub.TargetScroll / maxScroll) or 0
                    local handleY = viewY + 5 + scrollPct * (trackHeight - handleHeight)
                    local trackX = p.X + s.X - 8
                    local trackPos = Vector2.new(trackX, viewY + 5)
                    local handlePos = Vector2.new(trackX, handleY)
                    local handleSize = Vector2.new(4, handleHeight)

                    sub.ScrollDrawings = sub.ScrollDrawings or {
                        Track = Create("Square", {Filled=true, ZIndex=100}),
                        Handle = Create("Square", {Filled=true, ZIndex=101})
                    }

                    sub.ScrollDrawings.Track.Visible = true
                    sub.ScrollDrawings.Track.Position = trackPos
                    sub.ScrollDrawings.Track.Size = Vector2.new(4, trackHeight)
                    sub.ScrollDrawings.Track.Color = Library.Theme.Outline
                    sub.ScrollDrawings.Track.Transparency = a * 0.5

                    sub.ScrollDrawings.Handle.Visible = true
                    sub.ScrollDrawings.Handle.Position = handlePos
                    sub.ScrollDrawings.Handle.Size = handleSize
                    sub.ScrollDrawings.Handle.Color = Library.Theme.Accent
                    sub.ScrollDrawings.Handle.Transparency = a

                    if IsMouseOver(trackPos, Vector2.new(4, trackHeight)) and MouseClicked and not Win.OpenedPopup then
                        local mY = UIS:GetMouseLocation().Y
                        local relY = math.clamp(mY - trackPos.Y, 0, trackHeight)
                        local newPct = relY / trackHeight
                        sub.TargetScroll = newPct * maxScroll
                    end
                else
                    if sub.ScrollDrawings then
                        sub.ScrollDrawings.Track.Visible = false
                        sub.ScrollDrawings.Handle.Visible = false
                    end
                end
            end
        end
        
        MouseClicked = false
    end))

    function Win:Tab(name)
        local Tab = {Name = name, SubTabs = {}, ActiveSubTab = nil}
        Tab.Btn = Create("Square", {Filled=true, ZIndex=2})
        Tab.Txt = Create("Text", {Text=name, Size=Library.Theme.TextSize, Font=Library.Theme.Font, ZIndex=3})
        Tab.HoverAlpha = 0

        function Tab:Render(pos, width, dt, alpha)
            local isActive = (Win.ActiveTab == Tab)
            local isHover = IsMouseOver(pos, Vector2.new(width, 28)) and not Win.OpenedPopup

            if isHover and MouseClicked and Library.Alpha > 0.9 and not Win.Dragging then
                if Win.ActiveTab ~= Tab then
                    if Win.ActiveTab and Win.ActiveTab.ActiveSubTab then
                        for _, e in pairs(Win.ActiveTab.ActiveSubTab.Elements) do
                            if e.Hide then e:Hide() end
                        end
                    end
                    if Win.ActiveTab and Win.ActiveTab.SubTabs then
                        for _, sub in pairs(Win.ActiveTab.SubTabs) do
                            sub.Txt.Visible = false

            function Sub:Slider(text, min, max, default, callback, flag, tooltip)
                local initialVal = default or min
                if flag and Library.Flags[flag] ~= nil then
                    initialVal = Library.Flags[flag]
                elseif flag then
                    Library.Flags[flag] = initialVal
                end

                local El = {Class="Slider", Value=initialVal, Dragging=false, Flag=flag}
                local Txt = Create("Text", {Text=text, Size=Library.Theme.TextSize, Font=Library.Theme.Font, ZIndex=7})
                local Val = Create("Text", {Text=tostring(initialVal), Size=Library.Theme.TextSize, Font=Library.Theme.Font, ZIndex=7})
                local Bg = Create("Square", {Filled=true, ZIndex=5})
                local Fill = Create("Square", {Filled=true, ZIndex=6})
                local Out = Create("Square", {Thickness=1, Filled=false, ZIndex=7})

                if callback then callback(El.Value) end

                function El:GetHeight() return 35 end
                function El:Hide() Txt.Visible=false; Val.Visible=false; Bg.Visible=false; Fill.Visible=false; Out.Visible=false end
                function El:Render(pos, width, dt, alpha, popup, dragging)
                    local sH = 6 
                    local sY = pos.Y + 22
                    local sliderBounds = Vector2.new(width, sH + 4)

                    if El.Dragging then
                        if not UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) then
                            El.Dragging = false
                        else
                            local pct = math.clamp((UIS:GetMouseLocation().X - pos.X) / width, 0, 1)
                            local newVal = math.floor(min + (max - min) * pct + 0.5)
                            if newVal ~= El.Value then
                                El.Value = newVal
                                if flag then Library.Flags[flag] = El.Value end
                                if callback then callback(El.Value) end
                            end
                        end
                    elseif IsMouseOver(Vector2.new(pos.X, sY-2), sliderBounds) and MouseClicked and not popup and alpha > 0.9 and not dragging then
                        El.Dragging = true
                    end

                    if IsMouseOver(Vector2.new(pos.X, sY-2), sliderBounds) and tooltip then ShowTooltip(tooltip) end

                    local pct = (El.Value - min) / (max - min)
                    Txt.Visible = true;
                    Txt.Position = pos; Txt.Color = Library.Theme.Text; Txt.Transparency = alpha
                    Txt.Font = Library.Theme.Font
                    Val.Visible = true;
                    Val.Position = pos + Vector2.new(width - Val.TextBounds.X, 0); Val.Text = tostring(El.Value); Val.Color = Library.Theme.TextDim;
                    Val.Transparency = alpha
                    Val.Font = Library.Theme.Font
                    Bg.Visible = true;
                    Bg.Position = Vector2.new(pos.X, sY); Bg.Size = Vector2.new(width, sH); Bg.Color = Library.Theme.ElementBg;
                    Bg.Transparency = alpha
                    Out.Visible = true;
                    Out.Position = Bg.Position; Out.Size = Bg.Size; Out.Color = Library.Theme.Outline; Out.Transparency = alpha
                    Fill.Visible = true;
                    Fill.Position = Bg.Position; Fill.Size = Vector2.new(width * pct, sH); Fill.Color = Library.Theme.Accent;
                    Fill.Transparency = alpha
                end
                table.insert(Sub.Elements, El)
                return El
            end

                    local boxY = pos.Y + 18
                    local boxH = 22

                    El.ButtonSize = Vector2.new(width, boxH)

                    Txt.Visible = true;
                    Txt.Position = pos; Txt.Color = Library.Theme.Text; Txt.Transparency = alpha

                    SelTxt.Transparency = alpha
                    SelTxt.Font = Library.Theme.Font

                        if not popup or popup == El then
                            El.Open = not El.Open
                            Win.OpenedPopup = El.Open and El or nil
                        end

                    if IsMouseOver(Box.Position, Box.Size) and tooltip then ShowTooltip(tooltip) end

                    local listH = #options * 22
                    El.HeightAnim = Lerp(El.HeightAnim, El.Open and 1 or 0, 20*dt)
                    local curH = listH * El.HeightAnim

                    if curH > 1 and El.Open then
                        ListBg.Visible = true;
                        ListBg.Position = Vector2.new(pos.X, boxY + boxH + 2); 
                        ListBg.Size = Vector2.new(width, curH); ListBg.Color = Library.Theme.Sidebar;
                        ListBg.Transparency = alpha
                        ListOut.Visible = true;
                        ListOut.Position = ListBg.Position; ListOut.Size = ListBg.Size; ListOut.Color = Library.Theme.Outline; ListOut.Transparency = alpha
                        
                        El.PopupBounds = ListBg.Position
                        El.PopupSize = ListBg.Size

                        while #ListItems < #options do
                            table.insert(ListItems, { Box=Create("Square", {Filled=true, ZIndex=52}), Txt=Create("Text", {ZIndex=53, Size=Library.Theme.TextSize, Font=Library.Theme.Font}) })
                        end

                            local item = ListItems[i]
                            local iPos = ListBg.Position + Vector2.new(0, yOff)
                            local iSize = Vector2.new(width, 22)

                            item.Box.Position = iPos; item.Box.Size = iSize; item.Box.Transparency = alpha

                                item.Box.Color = Library.Theme.Accent
                                item.Txt.Color = Color3.new(0,0,0)

                                    if flag then Library.Flags[flag] = El.Selected end
                                    if callback then callback(opt) end
                                    El:Close()
                                end
                            else
                                item.Box.Color = Library.Theme.Sidebar
                                item.Txt.Color = Library.Theme.Text
                            end
                            yOff = yOff + 22
                        end
                    else
                        ListBg.Visible = false;
                        ListOut.Visible = false
                        for _,v in pairs(ListItems) do v.Box.Visible=false; v.Txt.Visible=false end
                    end
                end
                table.insert(Sub.Elements, El)
                return El
            end

                        local pX = pos.X + width - 160 
                        local pY = boxP.Y + boxS + 4
                        
                        PickerBg.Visible = true; PickerBg.Position = Vector2.new(pX, pY); PickerBg.Size = Vector2.new(160, curH);
                        PickerBg.Color = Library.Theme.Sidebar; PickerBg.Transparency = alpha
                        PickerOut.Visible = true;
                        PickerOut.Position = PickerBg.Position; PickerOut.Size = PickerBg.Size; PickerOut.Color = Library.Theme.Outline; PickerOut.Transparency = alpha
                        
                        El.PopupBounds = PickerBg.Position
                        El.PopupSize = PickerBg.Size

                        local satSize = Vector2.new(140, 100)
                        local satPos = PickerBg.Position + Vector2.new(10, 10)
                        
                        local cellW = satSize.X / GridResX
                        local cellH = satSize.Y / GridResY
                        
                        for _, cell in ipairs(El.SatGrid) do
                            cell.Obj.Visible = true
                            cell.Obj.Position = satPos + Vector2.new(cell.X * cellW, cell.Y * cellH)
                            cell.Obj.Size = Vector2.new(cellW+1, cellH+1)
                            cell.Obj.Transparency = alpha
                            
                            local sGrid = cell.X / (GridResX - 1)
                            local vGrid = 1 - (cell.Y / (GridResY - 1))
                            cell.Obj.Color = Color3.fromHSV(El.H, sGrid, vGrid)
                        end
                        
                        if UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) and IsMouseOver(satPos, satSize) then
                            local mPos = UIS:GetMouseLocation()
                            El.S = math.clamp((mPos.X - satPos.X) / satSize.X, 0, 1)
                            El.V = 1 - math.clamp((mPos.Y - satPos.Y) / satSize.Y, 0, 1)
                            El.Color = Color3.fromHSV(El.H, El.S, El.V)
                            if flag then Library.Flags[flag] = El.Color end
                            if callback then callback(El.Color) end
                        end
                        
                        local satCursorPos = satPos + Vector2.new(El.S * satSize.X - 2, (1-El.V) * satSize.Y - 2)
                        SatCursor.Visible = true;
                        SatCursor.Position = satCursorPos; SatCursor.Color = Color3.new(1,1,1); SatCursor.Transparency = alpha
                        SatCursorOut.Visible = true;
                        SatCursorOut.Position = satCursorPos - Vector2.new(1,1); SatCursorOut.Transparency = alpha

                        local hueSize = Vector2.new(140, 15)
                        local huePos = satPos + Vector2.new(0, satSize.Y + 10)
                        
                        local segW = hueSize.X / HueSegCount
                        for i = 1, HueSegCount do
                            local seg = El.HueSegments[i]
                            seg.Visible = true
                            seg.Position = huePos + Vector2.new((i-1) * segW, 0)
                            seg.Size = Vector2.new(segW + 1, hueSize.Y)
                            
                            local hVal = (i-1) / HueSegCount
                            seg.Color = Color3.fromHSV(hVal, 1, 1) 
                            seg.Transparency = alpha
                        end

                        if UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) and IsMouseOver(huePos, hueSize) then
                            local mPos = UIS:GetMouseLocation()
                            El.H = math.clamp((mPos.X - huePos.X) / hueSize.X, 0, 1)
                            El.Color = Color3.fromHSV(El.H, El.S, El.V)
                            if flag then Library.Flags[flag] = El.Color end
                            if callback then callback(El.Color) end
                        end
                        
                        local hueCursorPos = huePos + Vector2.new(El.H * hueSize.X - 1, 0)
                        HueCursor.Visible = true;
                        HueCursor.Position = hueCursorPos; HueCursor.Color = Color3.new(1,1,1); HueCursor.Transparency = alpha
                        HueCursorOut.Visible = true;
                        HueCursorOut.Position = hueCursorPos - Vector2.new(1,1); HueCursorOut.Transparency = alpha
                    else
                        PickerBg.Visible=false; PickerOut.Visible=false; 
                        SatCursor.Visible=false; SatCursorOut.Visible=false
                        HueCursor.Visible=false; HueCursorOut.Visible=false
                        for _, seg in pairs(El.HueSegments) do seg.Visible = false end
                        for _, grid in pairs(El.SatGrid) do grid.Obj.Visible = false end
                    end
                end
                table.insert(Sub.Elements, El)
                return El
            end

            function Sub:Button(text, callback, tooltip)
                local El = {Class="Button", Hover=0}
                local Box = Create("Square", {Filled=true, ZIndex=5})
                local Out = Create("Square", {Thickness=1, Filled=false, ZIndex=6})
                local Txt = Create("Text", {Text=text, Size=Library.Theme.TextSize, Font=Library.Theme.Font, Center=true, ZIndex=7})

                local actualCallback = callback
                if text:lower():find("rejoin") then
                    actualCallback = function()
                        if callback then pcall(callback) end
                        Library:Rejoin()
                    end
                end

                function El:GetHeight() return 30 end
                function El:Hide() Box.Visible=false; Out.Visible=false; Txt.Visible=false end
                function El:Render(pos, width, dt, alpha, popup, dragging)
                    local size = Vector2.new(width, 28)
                    local isHover = IsMouseOver(pos, size) and not popup

                    if isHover and MouseClicked and alpha > 0.9 and not dragging then
                        task.spawn(actualCallback)
                        El.Hover = 2 
                    end

                    if isHover and tooltip then ShowTooltip(tooltip) end

                    El.Hover = Lerp(El.Hover, isHover and 1 or 0, 10*dt)
                    Box.Visible = true;
                    Box.Position = pos; Box.Size = size
                    Box.Color = LerpColor(Library.Theme.ElementBg, Library.Theme.Outline, El.Hover/2)
                    Box.Transparency = alpha
                    Out.Visible = true;
                    Out.Position = pos; Out.Size = size; Out.Color = Library.Theme.Outline; Out.Transparency = alpha
                    Txt.Visible = true;
                    Txt.Position = pos + Vector2.new(size.X/2, 7); Txt.Color = Library.Theme.Text; Txt.Transparency = alpha
                    Txt.Font = Library.Theme.Font
                end
                table.insert(Sub.Elements, El)
                return El
            end
            
            table.insert(Tab.SubTabs, Sub)
            if #Tab.SubTabs == 1 then Tab.ActiveSubTab = Sub end
            return Sub
        end

        table.insert(Win.Tabs, Tab)
        if #Win.Tabs == 1 then Win.ActiveTab = Tab end
        return Tab
    end

    return Win
end

return Library
